namespace = SAIR_operation

#first event
espionage_operation_event = {
	id = SAIR_operation.100
	title = SAIR_operation.100.name
	desc = SAIR_operation.100.desc
	espionage_operation = yes
	picture = GFX_evt_space_station
	show_sound = event_default
	is_triggered_only = yes

	immediate = {#cheks if target lost al robots
		set_espionage_operation_progress_locked = yes
		if = {
			limit = {
				target = {
					NOT = {
						any_owned_planet = {
							any_owned_pop_group = {
								has_trait = trait_mechanical
							}
						}
					}
				}
			}
			set_espionage_operation_flag = operation_failed
		}
	}

	option = {#continue
		name = ACKNOWLEDGED
		trigger = {
			NOT = { has_espionage_operation_flag = operation_failed }
		}
		set_espionage_operation_progress_locked = no
	}

	option = {#target lost al robots
		name = ACKNOWLEDGED
		trigger = {
			has_espionage_operation_flag = operation_failed
		}
		owner = {
			country_event = { id = SAIR_operation.105 }
		}
	}
}

#syndaw.1000 but cut out trigger
country_event = {
	id = SAIR_operation.200
	hide_window = yes

	is_triggered_only = yes

	trigger = {	}

	immediate = {
		# uprising on colony
		if = {
			limit = {
				any_owned_planet = {
					any_owned_pop_group = {
						has_trait = trait_mechanical
					}
					is_capital = no
				}
			}
			random_owned_planet = {
				limit = {
					any_owned_pop_group = {
						has_trait = trait_mechanical
					}
					is_capital = no
				}
				carrier_event = { id = syndaw.1020 }
			}
		}
		# uprising wherever
		else = {
			random_owned_planet = {
				limit = {
					any_owned_pop_group = {
						has_trait = trait_mechanical
					}
				}
				carrier_event = { id = syndaw.1020 }
			}
		}
	}
}
